\chapter{Các phương pháp cải thiện (Improvement Methods)}
\label{chap:improvements}

\section{Phương pháp 1: Giới hạn độ sâu tối đa (Max Depth)}
\begin{itemize}
    \item \textbf{WHAT:} Thiết lập tham số \texttt{max\_depth = 4} để giới hạn cây không được vượt quá độ sâu 4 tầng (thay vì 7 tầng ở baseline).
    \item \textbf{RESULT:} Test Accuracy tăng từ 91.23\% lên 93.86\%, cấu trúc cây tinh gọn hơn rõ rệt với số lá giảm từ 19 xuống 11.
    \item \textbf{WHY:} Việc giới hạn độ sâu ép cây dừng phân nhánh sớm, ngăn chặn việc tạo ra các node lá mang quá ít mẫu dữ liệu. Điều này giúp loại bỏ sự phụ thuộc vào nhiễu nhỏ nhặt, cải thiện đáng kể khả năng tổng quát hóa (generalization) của mô hình.
\end{itemize}

\section{Phương pháp 2: Số lượng mẫu tối thiểu ở node lá (Min Samples Leaf)}
\begin{itemize}
    \item \textbf{WHAT:} Sử dụng tham số \texttt{min\_samples\_leaf = 3}, yêu cầu mỗi lá phải chứa ít nhất 3 mẫu, thay vì mặc định.
    \item \textbf{RESULT:} Test Accuracy duy trì mức 91.23\%, Train Accuracy giảm còn 98.02\%. Cấu trúc cây có sự thu gọn nhất định, giảm xuống còn 6 tầng và 13 lá.
    \item \textbf{WHY:} Buộc mỗi node lá đại diện cho nhiều hơn 1 mẫu giúp làm mượt cây quyết định. Mặc dù nó giúp cây bớt phức tạp hơn một chút, hiệu suất mang lại trên tập test vẫn chưa cải thiện đáng kể so với baseline trong trường hợp này.
\end{itemize}

\section{Phương pháp 3: Cắt tỉa (Pruning - Cost Complexity Pruning)}
\begin{itemize}
    \item \textbf{WHAT:} Áp dụng cắt tỉa với ngưỡng \texttt{ccp\_alpha = 0.005934} để loại bỏ đi các nhánh có ít khả năng mang lại thông tin hữu ích nhưng lại làm tăng độ phức tạp.
    \item \textbf{RESULT:} Test Accuracy nhảy vọt lên mức 93.86\%. Cấu trúc cây trở nên vô cùng tối giản với độ sâu chỉ vỏn vẹn 3 tầng và 6 lá.
    \item \textbf{WHY:} Phương pháp pruning loại bỏ trực tiếp các phần phân nhánh không mang lại giá trị gia tăng đáng kể đối với tập validation. Nhờ đó, nó mang lại một cấu trúc cây đơn giản nhất có thể mà vẫn duy trì độ chính xác cao nhất (ngăn ngừa overfitting triệt để).
\end{itemize}
